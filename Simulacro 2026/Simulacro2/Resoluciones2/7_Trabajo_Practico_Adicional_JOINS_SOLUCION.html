Resolución de los ejercicios
01

Ejercicio 1. Libros y sus autores

SELECT l.titulo, l.genero, l.anio_publicacion,
       a.nombre AS nombre_autor,
       a.apellido AS apellido_autor,
       a.nacionalidad
FROM libros l
INNER JOIN autores a ON l.id_autor = a.id_autor
WHERE l.anio_publicacion < 1980
ORDER BY l.anio_publicacion ASC;
02

Ejercicio 2. Préstamos realizados

SELECT p.id_prestamo,
       CONCAT(s.nombre, ' ', s.apellido) AS socio,
       l.titulo,
       p.fecha_prestamo,
       p.fecha_devolucion,
       p.estado
FROM libros l
INNER JOIN prestamos p ON l.id_libro = p.id_libro
INNER JOIN socios s ON p.id_socio = s.id_socio
ORDER BY p.fecha_prestamo DESC;
03

Ejercicio 3. Préstamos de determinados géneros

SELECT s.nombre AS nombre_socio,
       s.apellido AS apellido_socio,
       l.titulo,
       l.genero,
       a.nombre AS nombre_autor,
       a.apellido AS apellido_autor,
       p.fecha_prestamo,
       p.estado
FROM socios s
INNER JOIN prestamos p ON s.id_socio = p.id_socio
INNER JOIN libros l ON p.id_libro = l.id_libro
INNER JOIN autores a ON l.id_autor = a.id_autor
WHERE l.genero IN ('Novela', 'Distopia')
ORDER BY p.fecha_prestamo DESC
LIMIT 5;

Ejercicio 4. Préstamos recientes

SELECT s.nombre AS nombre_socio,
       s.apellido AS apellido_socio,
       l.titulo AS libro,
       a.nombre AS nombre_autor,
       a.apellido AS apellido_autor,
       p.fecha_prestamo,
       p.estado
FROM socios s
INNER JOIN prestamos p ON s.id_socio = p.id_socio
INNER JOIN libros l ON p.id_libro = l.id_libro
INNER JOIN autores a ON l.id_autor = a.id_autor
WHERE p.fecha_prestamo >= DATE_SUB(CURDATE(), INTERVAL 30 DAY)
ORDER BY p.fecha_prestamo DESC;

Ejercicio 5. Todos los socios y sus préstamos

SELECT CONCAT(s.nombre, ' ', s.apellido) AS socio,
       s.email,
       s.estado AS estado_socio,
       l.titulo,
       p.fecha_prestamo,
       p.estado AS estado_prestamo
FROM socios s
LEFT JOIN prestamos p ON s.id_socio = p.id_socio
LEFT JOIN libros l ON p.id_libro = l.id_libro
ORDER BY s.apellido, s.nombre;

Ejercicio 6. Autores y libros publicados

SELECT CONCAT(a.nombre, ' ', a.apellido) AS autor,
       a.nacionalidad,
       COALESCE(l.titulo, 'Sin libros cargados') AS titulo,
       COALESCE(l.genero, 'Sin género') AS genero
FROM autores a
LEFT JOIN libros l ON a.id_autor = l.id_autor
ORDER BY a.apellido, a.nombre;

Ejercicio 7. Libros nunca prestados

SELECT l.titulo,
       l.genero,
       l.anio_publicacion,
       CONCAT(a.nombre, ' ', a.apellido) AS autor,
       l.disponible
FROM libros l
LEFT JOIN prestamos p ON l.id_libro = p.id_libro
INNER JOIN autores a ON l.id_autor = a.id_autor
WHERE p.id_prestamo IS NULL
ORDER BY l.titulo;

Ejercicio 8. Préstamos pendientes

SELECT CONCAT(s.nombre, ' ', s.apellido) AS socio,
       s.email,
       l.titulo AS libro,
       CONCAT(a.nombre, ' ', a.apellido) AS autor,
       p.fecha_prestamo,
       p.estado,
       DATEDIFF(CURDATE(), p.fecha_prestamo) AS dias_transcurridos
FROM prestamos p
INNER JOIN socios s ON p.id_socio = s.id_socio
INNER JOIN libros l ON p.id_libro = l.id_libro
INNER JOIN autores a ON l.id_autor = a.id_autor
WHERE s.estado = 'Activo'
  AND p.estado IN ('Activo', 'Vencido')
  AND DATEDIFF(CURDATE(), p.fecha_prestamo) > 15
ORDER BY dias_transcurridos DESC;
Nota sobre el orden de ejecución: debido al orden lógico de ejecución de MySQL, la cláusula WHERE se evalúa antes que el SELECT. Por este motivo, en el WHERE todavía no está disponible el alias dias_transcurridos y es necesario repetir la expresión DATEDIFF(...). En cambio, ORDER BY se evalúa después del SELECT, por lo que allí sí puede utilizarse el alias.

Ejercicio 9. Autores vinculados a préstamos

SELECT DISTINCT
       CONCAT(a.nombre, ' ', a.apellido) AS autor,
       a.nacionalidad
FROM autores a
INNER JOIN libros l ON a.id_autor = l.id_autor
INNER JOIN prestamos p ON l.id_libro = p.id_libro
INNER JOIN socios s ON p.id_socio = s.id_socio
WHERE s.estado = 'Activo'
ORDER BY autor;

Ejercicio 10. Búsqueda de libros disponibles

SELECT l.titulo,
       l.genero,
       l.anio_publicacion,
       CONCAT(a.nombre, ' ', a.apellido) AS autor
FROM libros l
INNER JOIN autores a ON l.id_autor = a.id_autor
WHERE l.disponible = TRUE
  AND l.genero = 'Novela'
  AND l.titulo LIKE '%a%'
ORDER BY l.anio_publicacion DESC
LIMIT 3;
