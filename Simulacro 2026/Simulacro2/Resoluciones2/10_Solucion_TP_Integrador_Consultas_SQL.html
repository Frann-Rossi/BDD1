Ejercicio 1

SELECT p.nombre_producto, p.marca, c.nombre_categoria AS categoria, p.precio, p.stock
FROM productos p
INNER JOIN categorias c ON p.categoria_id = c.categoria_id
WHERE p.activo_producto = TRUE 
  AND p.stock > 0
  AND p.precio BETWEEN 25000 AND 150000
ORDER BY c.nombre_categoria, p.precio DESC;

Ejercicio 2

SELECT p.pedido_id, p.fecha_pedido, c.nombre_cliente, c.apellido_cliente, p.estado, p.canal
FROM clientes c
INNER JOIN pedidos p ON c.cliente_id = p.cliente_id
WHERE p.fecha_pedido BETWEEN '2026-01-01' AND '2026-06-30'
  AND p.estado IN ('Pagado', 'Pendiente')
ORDER BY p.fecha_pedido DESC;

Ejercicio 3

SELECT p.pedido_id, p.fecha_pedido,
       CONCAT(c.nombre_cliente, ' ', c.apellido_cliente) AS cliente, 
       pr.nombre_producto AS producto,
       ca.nombre_categoria AS categoria,
       dp.cantidad,
       pr.precio,
       (pr.precio * dp.cantidad) AS importe
FROM pedidos p
INNER JOIN clientes c ON p.cliente_id = c.cliente_id
INNER JOIN detalles_pedido dp ON p.pedido_id = dp.pedido_id
INNER JOIN productos pr ON dp.producto_id = pr.producto_id
INNER JOIN categorias ca ON pr.categoria_id = ca.categoria_id
WHERE p.canal = 'Web'
  AND p.fecha_pedido BETWEEN '2026-06-01' AND '2026-08-31'
  AND p.estado <> 'Cancelado'
  AND (pr.precio * dp.cantidad) > 100000;

Ejercicio 4

SELECT COUNT(DISTINCT p.pedido_id) AS cantidad_pedidos,
       SUM(dp.cantidad) AS cantidad_unidades,
       SUM(dp.cantidad * pr.precio) AS importe_total,
       MIN(pr.precio) AS menor_precio,
       MAX(pr.precio) AS mayor_precio
FROM pedidos p
INNER JOIN detalles_pedido dp ON p.pedido_id = dp.pedido_id
INNER JOIN productos pr ON dp.producto_id = pr.producto_id 
WHERE p.estado = 'Pagado'
  AND YEAR(p.fecha_pedido) = 2026;

Ejercicio 5

SELECT p.estado,
       COUNT(DISTINCT p.pedido_id) AS cantidad_pedidos,
       SUM(dp.cantidad) AS cantidad,
       SUM(pr.precio * dp.cantidad) AS importe
FROM pedidos p 
INNER JOIN detalles_pedido dp ON p.pedido_id = dp.pedido_id
INNER JOIN productos pr ON dp.producto_id = pr.producto_id
WHERE YEAR(p.fecha_pedido) = 2026
GROUP BY p.estado
ORDER BY importe DESC;

Ejercicio 6

SELECT MONTH(p.fecha_pedido) AS mes,
       COUNT(DISTINCT p.pedido_id) AS cant_pedidos,
       SUM(dp.cantidad) AS cantidad_unidades,
       SUM(pr.precio * dp.cantidad) AS importe
FROM pedidos p
INNER JOIN detalles_pedido dp ON p.pedido_id = dp.pedido_id
INNER JOIN productos pr ON dp.producto_id = pr.producto_id
WHERE p.estado = 'Pagado' 
  AND YEAR(p.fecha_pedido) = 2026
GROUP BY mes
ORDER BY mes;

Ejercicio 7

SELECT c.nombre_cliente,
       c.apellido_cliente,
       COUNT(p.pedido_id) AS cant_pedidos,
       MAX(p.fecha_pedido) AS fecha_ultimo_pedido,
       DATEDIFF(CURDATE(), MAX(p.fecha_pedido)) AS dias_transcurridos
FROM pedidos p
INNER JOIN clientes c ON p.cliente_id = c.cliente_id 
WHERE p.estado = 'Pagado'
  AND YEAR(p.fecha_pedido) = 2026
GROUP BY c.cliente_id, c.nombre_cliente, c.apellido_cliente
HAVING cant_pedidos >= 2
ORDER BY dias_transcurridos DESC;

Ejercicio 8

SELECT p.canal,
       p.estado,
       COUNT(DISTINCT p.pedido_id) AS cant_pedidos, 
       SUM(dp.cantidad) AS cantidad_unidades,
       SUM(pr.precio * dp.cantidad) AS total
FROM pedidos p 
INNER JOIN detalles_pedido dp ON p.pedido_id = dp.pedido_id
INNER JOIN productos pr ON dp.producto_id = pr.producto_id
WHERE p.fecha_pedido >= DATE_SUB(CURDATE(), INTERVAL 3 MONTH)
GROUP BY p.canal, p.estado
HAVING cant_pedidos >= 2;

Ejercicio 9

SELECT c.nombre_categoria,
       SUM(dp.cantidad) AS cantidad_unidades,
       SUM(pr.precio * dp.cantidad) AS importe
FROM categorias c
INNER JOIN productos pr ON c.categoria_id = pr.categoria_id
INNER JOIN detalles_pedido dp ON pr.producto_id = dp.producto_id
INNER JOIN pedidos pe ON dp.pedido_id = pe.pedido_id
WHERE pe.estado = 'Pagado' 
  AND YEAR(pe.fecha_pedido) = 2026
GROUP BY c.nombre_categoria
HAVING cantidad_unidades > 10 
   AND importe > 300000
ORDER BY importe DESC;

Ejercicio 10

SELECT c.cliente_id,
       c.nombre_cliente,
       c.apellido_cliente,
       COUNT(DISTINCT p.pedido_id) AS cant_pedidos,
       COALESCE(SUM(dp.cantidad), 0) AS cantidad_unidades,
       MIN(p.fecha_pedido) AS fecha_primer_pedido,
       MAX(p.fecha_pedido) AS fecha_ultimo_pedido
FROM clientes c
LEFT JOIN pedidos p ON c.cliente_id = p.cliente_id
LEFT JOIN detalles_pedido dp ON p.pedido_id = dp.pedido_id
GROUP BY c.cliente_id, c.nombre_cliente, c.apellido_cliente
ORDER BY cant_pedidos DESC;

Ejercicio 11 Integrador

SELECT MONTH(p.fecha_pedido) AS mes,
       c.nombre_categoria,
       COUNT(DISTINCT p.pedido_id) AS cant_pedidos,
       SUM(dp.cantidad) AS cant_unidades,
       SUM(dp.cantidad * pr.precio) AS importe
FROM categorias c
INNER JOIN productos pr ON c.categoria_id = pr.categoria_id
INNER JOIN detalles_pedido dp ON pr.producto_id = dp.producto_id
INNER JOIN pedidos p ON dp.pedido_id = p.pedido_id
WHERE p.estado = 'Pagado'
  AND YEAR(p.fecha_pedido) = 2026
  AND pr.activo_producto = TRUE
GROUP BY mes, c.nombre_categoria
HAVING cant_unidades > 10
   AND importe > 250000
ORDER BY mes, importe DESC;
